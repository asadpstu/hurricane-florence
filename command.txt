import pandas as pd

p = pd.read_csv(
    "output/flood_mapping/florence_2018_physics_v3_5_retained/daily_flood_summary.csv"

    "output/flood_mapping/florence_2018_ml_v4_retained/daily_flood_summary.csv"
)

if "state_type" in p.columns:
    p = p[p["state_type"] == "daily_max"].copy()

    m = m[m["state_type"] == "daily_max"].copy()

    [
        "date",
        "predicted_q_m3s",
        "predicted_gage_height_ft",
        "flood_area_km2",
    ]

    "predicted_q_m3s": "physics_q_m3s",
    "predicted_gage_height_ft": "physics_stage_ft",
    "flood_area_km2": "physics_flood_km2",
})

m = m[
    [
        "date",
        "predicted_q_m3s",
        "predicted_gage_height_ft",
        "flood_area_km2",

].rename(columns={
    "predicted_q_m3s": "ml_q_m3s",
    "predicted_gage_height_ft": "ml_stage_ft",

})
